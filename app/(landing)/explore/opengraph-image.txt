import { ImageResponse } from "next/og";
import { wordSearchService } from "@/api_functions/explore/word-search-service";
import { SortType, PreferredGender } from "@data/enums";

export const runtime = "edge";
export const alt = "Service Listings";
export const size = {
  width: 1200,
  height: 630,
};

export default async function Image() {
  // Fetch the first 6 services
  const explore = await wordSearchService({
    query: "",
    filter: {
      state: null,
      verified: true,
      online: true,
      sort: {
        range: SortType.HIGH_TO_LOW,
        type: "rating",
      },
      customerGender: PreferredGender.UNISEX,
      partnerGender: null,
    },
    category: null,
    pagination: {
      limit: 6,
      page: 1,
    },
  });

  if (!explore) {
    return new ImageResponse(<div>No services found</div>, {
      ...size,
    });
  }

  const serviceImages = explore.data.map(
    (service) => service.service.imageUrls[0]
  );

  const mostPopularServiceImage = serviceImages.sort(
    (a, b) => b.length - a.length
  )[0];

  //take the first 6 images
  const firstSixImages = serviceImages.slice(0, 6);

  return new ImageResponse(
    (
      <div
        style={{
          background: "linear-gradient(to bottom right, #4F46E5, #0EA5E9)",
          width: "100%",
          height: "100%",
          display: "flex",
          flexDirection: "column",
          alignItems: "center",
          justifyContent: "center",
          padding: "40px",
        }}
      >
        <div
          style={{
            display: "grid",
            gridTemplateColumns: "repeat(3, 1fr)",
            gap: "20px",
            marginBottom: "40px",
          }}
        >
          {explore?.data.map((service, index) => (
            <div
              key={index}
              style={{
                background: "white",
                borderRadius: "12px",
                padding: "15px",
                boxShadow: "0 4px 6px rgba(0, 0, 0, 0.1)",
                display: "flex",
                flexDirection: "column",
              }}
            >
              <img
                src={firstSixImages[index] || "/default-service.png"}
                alt={service.service.title}
                style={{
                  width: "100%",
                  height: "120px",
                  objectFit: "cover",
                  borderRadius: "8px",
                }}
              />
              <div style={{ marginTop: "10px" }}>
                <p
                  style={{
                    fontSize: "16px",
                    fontWeight: "bold",
                    margin: "0",
                    color: "#1F2937",
                  }}
                >
                  {service.service.title}
                </p>
                <p
                  style={{
                    fontSize: "14px",
                    margin: "4px 0 0 0",
                    color: "#6B7280",
                  }}
                >
                  ⭐{" "}
                  {service.service.rating
                    ? service.service.rating.toFixed(1)
                    : 0}
                </p>
              </div>
            </div>
          ))}
        </div>
        <div
          style={{
            textAlign: "center",
            color: "white",
          }}
        >
          <h1
            style={{
              fontSize: "48px",
              margin: "0",
              fontWeight: "bold",
            }}
          >
            Discover Top Services
          </h1>
          <p
            style={{
              fontSize: "24px",
              opacity: 0.9,
              margin: "16px 0 0 0",
            }}
          >
            Explore our highly-rated verified services
          </p>
        </div>
      </div>
    ),
    {
      ...size,
    }
  );
}
