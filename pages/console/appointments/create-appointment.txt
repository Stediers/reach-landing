import { createAppointment } from "@api_functions/appointment/create-appointment";
import { UserAutoCompleteResult } from "@api_functions/gig/fetch-user-autocomplete-result";
import fetchAddOns from "@api_functions/service/add-on/fetch-add-ons";
import {
  ServicesScreen,
  fetchServicesScreen,
} from "@api_functions/service/fetch-services-screen";
import { showOkPopup } from "@components/notifications/Popup";
import { showSnackBar } from "@components/notifications/Snackbar";
import { State } from "@data/enums";
import { AddOn, LocationAttributes, Slot, StepData } from "@data/types";
import { verifyMobileNumber } from "@helper_functions/verify";
import Desktop from "@src/console/appointments/create-appointment/Desktop";
import Mobile from "@src/console/appointments/create-appointment/Mobile";
import ConsoleWrapper from "@wrapper/ConsoleWrapper";
import { add } from "lodash";
import { GetServerSidePropsContext } from "next";
import router from "next/router";
import { Dispatch, SetStateAction, useEffect, useState } from "react";

export function getServerSideProps(context: GetServerSidePropsContext) {
  const date = context.query.date;

  if (!date || (date && typeof date !== "string")) {
    return {
      redirect: {
        destination: "/console/appointments",
        permanent: false,
      },
    };
  }

  return {
    props: {
      selectedDate: date ?? null,
    },
  };
}

export default function Main({ selectedDate }: { selectedDate: string }) {
  const [pageState, setPageState] = useState<State>(State.LOADING);
  const [selectedService, setSelectedService] = useState<ServicesScreen | null>(
    null
  );
  const [services, setServices] = useState<ServicesScreen[]>([]);
  const [buttonState, setButtonState] = useState<State>(State.SUCCESS);
  const [addOns, setAddOns] = useState<AddOn[]>([]);
  const [selectedAddOns, setSelectedAddOns] = useState<string[]>([]);

  useEffect(() => {
    fetchServicesScreen().then((res1) => {
      if (res1) {
        if (res1.servicesScreen.length === 0) {
          showOkPopup({
            message: "You need to create a service first",
            title: "No services found",
          }).then(() => {
            router.push("/console/services/add-service");
          });
        } else {
          fetchAddOns().then((res2) => {
            if (res2) {
              setAddOns(res2);
              setServices(res1.servicesScreen);
              setSelectedService(res1.servicesScreen[0]);
              setPageState(State.SUCCESS);
            }
          });
        }
      } else {
        setPageState(State.ERROR);
      }
    });
  }, []);

  const [mobileNumber, setMobileNumber] = useState<string>("");

  const [selectedUser, setSelectedUser] =
    useState<UserAutoCompleteResult | null>(null);

  const [date, setDate] = useState<Date | null>(
    selectedDate ? new Date(selectedDate) : new Date()
  );
  const [canSave, setCanSave] = useState<boolean>(false);

  const [slot, setSlot] = useState<Slot | null>(null);

  const [steps, setSteps] = useState<StepData[]>([
    {
      title: "Select Service",
      description: "Which service?",
      index: 0,
      state: verifyService(selectedService),
    },
    {
      title: "Select Slot",
      description: "When?",
      index: 1,
      state: verifySlot(slot),
    },
    {
      title: "Customer Details",
      description: "Who?",
      index: 2,
      state: verifyUser({ mobileNumber, selectedUser }),
    },
    {
      title: "Add-ons",
      description: "Add-ons",
      index: 3,
      state: true,
    },
    {
      title: "Review",
      description: "Review and confirm",
      index: 4,
      state: true,
    },
  ]);

  const [currentStep, setCurrentStep] = useState<StepData>(steps[0]);

  useEffect(() => {
    setSteps((steps) => {
      const newSteps = [...steps];
      newSteps[0].state = verifyService(selectedService);
      newSteps[1].state = verifySlot(slot);
      newSteps[2].state = verifyUser({ mobileNumber, selectedUser });
      return newSteps;
    });
  }, [selectedService, slot, mobileNumber, selectedUser, currentStep]);

  useEffect(() => {
    setSlot(null);
  }, [date]);

  return (
    <ConsoleWrapper
      title="Create Appointment"
      state={pageState}
      mobileJSX={
        <Mobile
          buttonState={buttonState}
          date={date}
          setDate={setDate}
          selectedUser={selectedUser}
          setSelectedUser={setSelectedUser}
          mobileNumber={mobileNumber}
          setMobileNumber={setMobileNumber}
          selectedService={selectedService}
          setSelectedService={setSelectedService}
          services={services}
          canSave={canSave}
          setButtonState={setButtonState}
          slot={slot}
          setSlot={setSlot}
          steps={steps}
          currentStep={currentStep}
          setCurrentStep={setCurrentStep}
          addOns={addOns}
          selectedAddOns={selectedAddOns}
          setSelectedAddOns={setSelectedAddOns}
          createAppointment={() =>
            createAppointmentCallback({
              mobileNumber: mobileNumber,
              selectedService: selectedService,
              selectedSlot: slot,
              selectedUser: selectedUser,
              setButtonState: setButtonState,
              steps: steps,
              addOnIds: selectedAddOns,
            })
          }
        />
      }
      desktopJSX={
        <Desktop
          buttonState={buttonState}
          date={date}
          setDate={setDate}
          selectedUser={selectedUser}
          setSelectedUser={setSelectedUser}
          mobileNumber={mobileNumber}
          setMobileNumber={setMobileNumber}
          selectedService={selectedService}
          setSelectedService={setSelectedService}
          services={services}
          canSave={canSave}
          setButtonState={setButtonState}
        />
      }
    />
  );
}

function verifyService(service: ServicesScreen | null) {
  if (!service) {
    return false;
  }
  return true;
}

function verifySlot(slot: Slot | null) {
  if (!slot) {
    return false;
  }
  return true;
}

function verifyUser({
  mobileNumber,
  selectedUser,
}: {
  mobileNumber: string;
  selectedUser: UserAutoCompleteResult | null;
}) {
  if (selectedUser) {
    if (!verifyMobileNumber(mobileNumber)) {
      return false;
    }
  } else if (!verifyMobileNumber(mobileNumber)) {
    return false;
  }

  return true;
}

async function createAppointmentCallback({
  setButtonState,
  selectedSlot,
  selectedService,
  selectedUser,
  mobileNumber,
  steps,
  addOnIds,
}: {
  setButtonState: Dispatch<SetStateAction<State>>;
  selectedSlot: Slot | null;
  selectedService: ServicesScreen | null;
  selectedUser: UserAutoCompleteResult | null;
  mobileNumber: string;
  steps: StepData[];
  addOnIds: string[];
}) {
  if (steps.some((step) => !step.state)) {
    showSnackBar({
      message: "Please complete all steps",
      state: State.ERROR,
    });
    return false;
  } else {
    setButtonState(State.LOADING);
    await createAppointment({
      date: selectedSlot!!.date,
      mobileNumber: mobileNumber,
      serviceId: selectedService!!.serviceId,
      userId: selectedUser ? selectedUser.userId : null,
      addOnIds: addOnIds,
    });
    setButtonState(State.SUCCESS);
    router.push("/console/appointments");
    return true;
  }
}
